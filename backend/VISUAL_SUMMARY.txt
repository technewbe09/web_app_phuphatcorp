┌─────────────────────────────────────────────────────────────────────────────┐
│ DATA FLOW: Request with Auth                                                │
├─────────────────────────────────────────────────────────────────────────────┤
│                                                                             │
│  1. HTTP Request arrives                                                    │
│     Authorization: Bearer eyJhbGc...                                        │
│                                                                             │
│  2. authenticateToken middleware                                            │
│     ├─ Extract token from header                                            │
│     ├─ Verify signature with JWT_SECRET                                    │
│     ├─ Decode: { userId, email, role, permissions[] }                      │
│     ├─ Attach to req.user                                                   │
│     └─ Query DB for role active status (async, fire-and-forget)            │
│                                                                             │
│  3. requirePermission('roles.view') middleware                              │
│     ├─ Check req.user exists                                                │
│     ├─ Check 'roles.view' in req.user.permissions array                    │
│     └─ If not found: return 403                                             │
│                                                                             │
│  4. Controller receives request                                             │
│     └─ Calls service layer                                                  │
│                                                                             │
│  5. Service executes business logic                                         │
│     ├─ Queries database (via pool)                                          │
│     ├─ Processes data                                                       │
│     ├─ Throws ServiceError if validation fails                              │
│     └─ Returns result                                                       │
│                                                                             │
│  6. Controller returns response                                             │
│     ├─ If error: catches ServiceError                                       │
│     ├─ Maps statusCode to HTTP status                                       │
│     └─ Returns JSON response                                                │
│                                                                             │
└─────────────────────────────────────────────────────────────────────────────┘

│ ERROR HANDLING PATTERN                                                      │
├─────────────────────────────────────────────────────────────────────────────┤
│                                                                             │
│  Service Layer (Throws):                                                    │
│  ┌──────────────────────────────────────────────────────────────┐          │
│  │ export class ServiceError extends Error {                    │          │
│  │   constructor(                                               │          │
│  │     message: string,                                         │          │
│  │     public statusCode: number,                               │          │
│  │     public code?: string                                     │          │
│  │   ) { super(message) }                                       │          │
│  │ }                                                            │          │
│  │                                                              │          │
│  │ throw new ServiceError('Not found', 404, 'NOT_FOUND')        │          │
│  │ throw new ServiceError('Duplicate', 409, 'DUPLICATE_CODE')   │          │
│  └──────────────────────────────────────────────────────────────┘          │
│                                                                             │
│  Controller Layer (Catches):                                                │
│  ┌──────────────────────────────────────────────────────────────┐          │

│  │   result = await service.method()                            │          │

│  │   if (err instanceof ServiceError) {                         │          │
│  │     sendError(res, err.message, err.statusCode, err.code)    │          │
│  │   } else {                                                   │          │
│  │     sendError(res, 'Server error', 500)                      │          │
│  │   }                                                          │          │
